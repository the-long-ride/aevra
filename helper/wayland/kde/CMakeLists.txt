cmake_minimum_required(VERSION 3.22)
project(aevra_wayland_kde VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Gui Test)
add_executable(policy-test policy_test.cpp)
target_link_libraries(policy-test PRIVATE Qt6::Core Qt6::Test)
enable_testing()
add_test(NAME policy COMMAND policy-test)
option(AEVRA_BUILD_PLUGIN "Build KWin native plugin" ON)
option(AEVRA_NATIVE_FIXTURE "Include isolated test-only controls; NEVER distribute" OFF)
if(AEVRA_BUILD_PLUGIN)
 find_package(Qt6 REQUIRED COMPONENTS Widgets Quick)
 find_package(KWin 6.6 REQUIRED)
 if(NOT KWin_VERSION VERSION_LESS 6.7)
  message(FATAL_ERROR "This adapter targets KWin 6.6; validate newer ABI before building")
 endif()
 add_library(aevra-desktop MODULE bridge.cpp device.cpp pixels.cpp)
 target_link_libraries(aevra-desktop PRIVATE KWin::kwin Qt6::DBus)
 target_compile_definitions(aevra-desktop PRIVATE $<$<BOOL:${AEVRA_NATIVE_FIXTURE}>:AEVRA_NATIVE_FIXTURE>)
 set_target_properties(aevra-desktop PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/kwin/plugins")
 install(TARGETS aevra-desktop DESTINATION lib/qt6/plugins/kwin/plugins)
endif()
